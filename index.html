<div class="header">

    <input type="text" id="newUrl" placeholder="الصق رابط يوتيوب (مثال: https://youtu.be/xxx)" />

    <button class="btn-add" onclick="addFrame()">➕ إضافة</button>

    <input type="number" id="loopTime" value="25" min="5" />

    <span style="font-size:13px;">ثانية</span>

    <button class="btn-play" id="playBtn" onclick="toggleLoop()">▶️ تشغيل تلقائي</button>

    <button class="btn-stop" onclick="stopAll()">⏹️ إيقاف الكل</button>

    <span class="status">🔄 الحالة: <span id="statusText" class="off">متوقف</span></span>

<div class="grid" id="gridContainer"></div>

<script>

    // ---------- قاعدة البيانات ----------

    let frames = JSON.parse(localStorage.getItem('myFrames')) || [];

    let loopInterval = null;

    let countDown = 0;

    // ---------- دالة تحويل رابط يوتيوب فقط (مضمونة) ----------

    function getYoutubeEmbed(url) {

        let u = url.trim();

        let match = u.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([^&\?\/]+)/);

        if (match) {

            return `https://www.youtube.com/embed/${match[1]}?autoplay=1&mute=1&playsinline=1`;

        }

        return null; // ليس يوتيوب

    }

    // ---------- إضافة إطار جديد ----------

    function addFrame() {

        const input = document.getElementById('newUrl');

        let url = input.value.trim();

        if (!url) return alert('⚠️ الرجاء إدخال رابط يوتيوب');

        // تحقق من صحة الرابط

        if (!getYoutubeEmbed(url)) {

            return alert('⚠️ هذا الرابط غير مدعوم. الرجاء استخدام رابط يوتيوب فقط (لضمان العمل).');

        }

        const newFrame = {

            id: 'f_' + Date.now(),

            url: url,

            title: '🎬 يوتيوب #' + (frames.length + 1),

            views: 0

        };

        frames.push(newFrame);

        input.value = '';

        saveAndRender();

    }

    // ---------- حذف إطار ----------

    function deleteFrame(id) {

        if (!confirm('حذف هذا الإطار؟')) return;

        frames = frames.filter(f => f.id !== id);

        if (loopInterval) toggleLoop(); // إيقاف التكرار إذا كان شغال

        saveAndRender();

    }

    // ---------- تحديث رابط إطار ----------

    function updateFrame(id) {

        const input = document.querySelector(`input[data-id="${id}"]`);

        let newUrl = input.value.trim();

        if (!newUrl) return alert('أدخل رابطاً');

       

        if (!getYoutubeEmbed(newUrl)) {

            return alert('⚠️ الرابط غير صحيح. استخدم يوتيوب فقط.');

        }

        const frame = frames.find(f => f.id === id);

        if (frame) {

            frame.url = newUrl;

            frame.title = '🎬 يوتيوب #' + (frames.indexOf(frame) + 1);

            saveAndRender();

            // لو كان التكرار شغال، نحدث الإطار فوراً

            if (loopInterval) {

                const iframe = document.getElementById(`if_${frame.id}`);

                if (iframe) iframe.src = getYoutubeEmbed(newUrl);

            }

        }

    }

    // ---------- تشغيل جميع الإطارات (مع زيادة المشاهدات) ----------

    function reloadAll() {

        if (frames.length === 0) {

            stopLoop();

            return;

        }

        // 1. زيادة المشاهدات لكل الإطارات

        frames.forEach(f => f.views = (f.views || 0) + 1);

        // 2. تحديث مصدر الـ iframes مع تفعيل التشغيل التلقائي

        frames.forEach(f => {

            const iframe = document.getElementById(`if_${f.id}`);

            if (iframe) {

                iframe.src = getYoutubeEmbed(f.url);

            }

        });

        // 3. حفظ وتحديث الواجهة

        saveAndRender();

        // 4. تحديث المؤقت البصري

        const duration = parseInt(document.getElementById('loopTime').value) || 25;

        countDown = duration;

        document.getElementById('statusText').textContent = `يعمل (${countDown}ث)`;

        document.getElementById('statusText').className = 'on';

    }

    // ---------- بدء حلقة التكرار ----------

    function startLoop() {

        if (loopInterval) return;

        if (frames.length === 0) return alert('⚠️ أضف فيديوهات أولاً!');

        const duration = parseInt(document.getElementById('loopTime').value) || 25;

        if (duration < 3) return alert('المدة قصيرة جداً (أقل من 3 ثواني)');

        // نشغل أول مرة فوراً

        reloadAll();

        // نبدأ المؤقت

        loopInterval = setInterval(() => {

            reloadAll();

        }, duration * 1000);

        // تغيير شكل الزر

        const btn = document.getElementById('playBtn');

        btn.textContent = '⏹️ إيقاف التكرار';

        btn.classList.add('active');

    }

    // ---------- إيقاف حلقة التكرار ----------

    function stopLoop() {

        if (loopInterval) {

            clearInterval(loopInterval);

            loopInterval = null;

        }

        const btn = document.getElementById('playBtn');

        btn.textContent = '▶️ تشغيل تلقائي';

        btn.classList.remove('active');

        document.getElementById('statusText').textContent = 'متوقف';

        document.getElementById('statusText').className = 'off';

        document.getElementById('statusText').textContent = 'متوقف';

    }

    // ---------- تبديل التشغيل التلقائي ----------

    function toggleLoop() {

        if (loopInterval) {

            stopLoop();

        } else {

            startLoop();

        }

    }

    // ---------- إيقاف الكل (تفريغ الإطارات) ----------

    function stopAll() {

        stopLoop();

        frames.forEach(f => {

            const iframe = document.getElementById(`if_${f.id}`);

            if (iframe) iframe.src = '';

        });

        document.getElementById('statusText').textContent = 'متوقف';

        document.getElementById('statusText').className = 'off';

    }

    // ---------- حفظ وعرض الشبكة ----------

    function saveAndRender() {

        localStorage.setItem('myFrames', JSON.stringify(frames));

        const grid = document.getElementById('gridContainer');

        grid.innerHTML = '';

        if (frames.length === 0) {

            grid.innerHTML = `<div class="empty">📭 لا توجد إطارات. أضف رابط يوتيوب في الأعلى</div>`;

            return;

        }

        frames.forEach(f => {

            const card = document.createElement('div');

            card.className = 'card';

            // نضع الرابط بدون تشغيل تلقائي في البداية (لكن عند الضغط على "تشغيل" يصبح تلقائي)

            const embedSrc = getYoutubeEmbed(f.url) || '';

            card.innerHTML = `

                <div class="video-box">

                    <iframe id="if_${f.id}" src="${embedSrc}" allow="autoplay; encrypted-media; fullscreen" allowfullscreen></iframe>

                </div>

                <div class="info-bar">

                    <span class="title">${f.title}</span>

                    <span class="views">👁️ <span id="view_${f.id}">${f.views || 0}</span></span>

                </div>

                <div class="actions">

                    <input type="text" data-id="${f.id}" placeholder="تعديل الرابط..." value="${f.url}" />

                    <button class="btn-update" onclick="updateFrame('${f.id}')">تحديث</button>

                    <button class="btn-del" onclick="deleteFrame('${f.id}')">✖</button>

                </div>

            `;

            grid.appendChild(card);

        });

    }

    // ---------- عد تنازلي بصري كل ثانية ----------

    setInterval(() => {

        if (loopInterval) {

            if (countDown > 0) {

                countDown--;

                document.getElementById('statusText').textContent = `يعمل (${countDown}ث)`;

            }

        }

    }, 1000);

    // ---------- عند تحميل الصفحة ----------

    window.onload = function() {

        saveAndRender();

        stopLoop(); // نوقف أي مؤقت عالق

    };

    // ---------- إضافة بالضغط على Enter ----------

    document.getElementById('newUrl').addEventListener('keypress', function(e) {

        if (e.key === 'Enter') addFrame();

    });

</script>
